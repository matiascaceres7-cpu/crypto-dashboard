import streamlit as st
import os
import requests
import pandas as pd
from dotenv import load_dotenv

# Cargar la API Key desde el archivo .env
load_dotenv()
API_KEY = os.getenv("COINGECKO_API_KEY")

# URL base para la API de CoinGecko
BASE_URL = "https://api.coingecko.com/api/v3"

# Headers obligatorios para la API Demo
HEADERS = {"accept": "application/json"}
if API_KEY:
    HEADERS["x-cg-demo-api-key"] = API_KEY


@st.cache_data(ttl=60)
def obtener_top_criptos():
    url = f"{BASE_URL}/coins/markets"
    params = {
        "vs_currency": "usd",
        "order": "market_cap_desc",
        "per_page": 250,
        "page": 1,
        "sparkline": "false",
    }

    try:
        response = requests.get(url, headers=HEADERS, params=params, timeout=15)
        if response.status_code == 200:
            datos = response.json()
            df = pd.DataFrame(datos)
            columnas_interesantes = ['id', 'name', 'symbol', 'current_price', 'market_cap', 'price_change_percentage_24h', 'image']
            return df[columnas_interesantes]
        print(f"Error {response.status_code}: {response.text}")
        return None
    except Exception as e:
        print(f"Error de conexión: {e}")
        return None


def obtener_historico_activo(coin_id="bitcoin", days=30):
    """
    Consume el endpoint de rango histórico de CoinGecko.
    Retorna un DataFrame con Fecha y Precio.
    """
    url = f"{BASE_URL}/coins/{coin_id}/market_chart"
    params = {
        "vs_currency": "usd",
        "days": days,
        "interval": "daily",
    }
    try:
        response = requests.get(url, headers=HEADERS, params=params, timeout=15)
        if response.status_code == 200:
            precios = response.json().get('prices', [])
            if not precios:
                return None
            df_hist = pd.DataFrame(precios, columns=['Fecha', 'Precio'])
            df_hist['Fecha'] = pd.to_datetime(df_hist['Fecha'], unit='ms')
            return df_hist
        print(f"Error en API Histórica {response.status_code}: {response.text}")
        return None
    except Exception as e:
        print(f"Error de conexión en módulo histórico: {e}")
        return None


# Prueba rápida para verificar la conexión
if __name__ == "__main__":
    print("Probando conexión con CoinGecko...")
    df_prueba = obtener_top_criptos()
    if df_prueba is not None:
        print("\nConexión exitosa. Primeros 3 activos:")
        print(df_prueba.head(3))
